# dlsm-shm (C) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the `dlsm-shm` C library — a named, fixed-base shared-memory segment with an append-only atomic-bump arena allocator, postmaster-style stale recovery, and read-only observer attach — plus a CMake project and high-equivalent C verification (Unity/CTest, seeded-PRNG property tests, TSAN stress, libFuzzer, microbench, llvm-cov 100%).

**Architecture:** Single segment = `ShmHeader` + arena. Creator maps `MAP_SHARED|MAP_FIXED_NOREPLACE` at a fixed base (`0x100000000000`); observers re-map `MAP_FIXED, PROT_READ` at the base read from the header, so the engine's absolute pointers resolve. Allocator is a CAS bump pointer that never frees a single object. Recovery uses `kill(owner_pid, 0)` liveness and re-initializes the arena (durable state is rebuilt from WAL elsewhere).

**Tech Stack:** C11, CMake ≥ 3.20, Clang (for libFuzzer + llvm-cov), pthreads, POSIX `shm_open`/`mmap`, CPM.cmake for fetching Unity. Linux x86_64 only.

Design spec: `docs/superpowers/specs/2026-06-14-dlsm-shm-c-design.md`.

```c
#include <stdbool.h>

/* shm error band: 10000+ (architecture.md §8). */
enum {
    DLSM_SHM_E_OPEN          = 10001, /* shm_open failed */
    DLSM_SHM_E_FTRUNCATE     = 10002, /* ftruncate failed */
    DLSM_SHM_E_BASE_OCCUPIED = 10003, /* fixed base address unavailable */
    DLSM_SHM_E_IN_USE        = 10004, /* segment owned by a live process */
    DLSM_SHM_E_BAD_MAGIC     = 10005, /* header magic mismatch */
    DLSM_SHM_E_BAD_VERSION   = 10006, /* header version mismatch */
    DLSM_SHM_E_OOM           = 10007, /* arena exhausted */
    DLSM_SHM_E_INVAL         = 10008  /* invalid argument */
};

typedef struct dlsm_shm dlsm_shm; /* opaque */

/* Lifecycle (single-writer host). */
dlsm_status dlsm_shm_create_or_recover(const char *name, size_t size, dlsm_shm **out);
dlsm_status dlsm_shm_attach_readonly(const char *name, dlsm_shm **out);
void        dlsm_shm_detach(dlsm_shm *s);
dlsm_status dlsm_shm_cleanup_if_stale(const char *name, bool *cleaned);

/* Allocation: CAS bump, multi-thread safe, returns an absolute pointer
 * inside the segment, aligned to `align` (power of two). Returns NULL on
 * exhaustion / invalid args / read-only handle. Never frees a single object. */
void *dlsm_shm_alloc(dlsm_shm *s, size_t size, size_t align);

/* Introspection. */
void  *dlsm_shm_base(const dlsm_shm *s);
size_t dlsm_shm_used(const dlsm_shm *s);
size_t dlsm_shm_capacity(const dlsm_shm *s);

/* English message for an shm-band status. */
const char *dlsm_shm_strerror(dlsm_status st);

#endif /* DLSM_SHM_H */
```

- [ ] **Step 2: Add the full implementation skeleton to `libs/shm/src/shm.c`**

Replace the entire contents of `libs/shm/src/shm.c` with (keeps the strerror table, adds the create path + helpers; later tasks fill more):
```c
#define _GNU_SOURCE
#include "dlsm/shm.h"

#include <errno.h>
#include <fcntl.h>
#include <signal.h>
#include <stdatomic.h>
#include <stdint.h>
#include <stdlib.h>
#include <string.h>
#include <sys/mman.h>
#include <unistd.h>

#define DLSM_SHM_MAGIC   0x4D4853204D534C44ULL /* "DLSM SHM" little-endian-ish */
#define DLSM_SHM_VERSION 1u
#ifndef DLSM_SHM_BASE_ADDR
#define DLSM_SHM_BASE_ADDR 0x100000000000ULL /* 16 TiB */
#endif

struct ShmHeader {
    uint64_t magic;
    uint32_t version;
    uint32_t _pad;
    uint64_t base_addr;
    uint64_t total_size;
    _Atomic uint64_t bump;
    int32_t  owner_pid;
};

struct dlsm_shm {
    struct ShmHeader *hdr; /* == mapped base */
    size_t mapped_size;
    int    fd;
    bool   readonly;
};

static size_t align_up(size_t v, size_t a) { return (v + (a - 1)) & ~(a - 1); }

static void init_header(struct ShmHeader *h, size_t size) {
    h->magic      = DLSM_SHM_MAGIC;
    h->version    = DLSM_SHM_VERSION;
    h->_pad       = 0;
    h->base_addr  = (uint64_t)DLSM_SHM_BASE_ADDR;
    h->total_size = (uint64_t)size;
    atomic_store_explicit(&h->bump,
        align_up(sizeof(struct ShmHeader), 16), memory_order_relaxed);
    h->owner_pid  = (int32_t)getpid();
}

const char *dlsm_shm_strerror(dlsm_status st) {
    switch (st) {
    case DLSM_SHM_E_OPEN:          return "shm_open failed";
    case DLSM_SHM_E_FTRUNCATE:     return "ftruncate failed";
    case DLSM_SHM_E_BASE_OCCUPIED: return "fixed base address occupied";
    case DLSM_SHM_E_IN_USE:        return "segment in use by a live owner";
    case DLSM_SHM_E_BAD_MAGIC:     return "bad segment magic";
    case DLSM_SHM_E_BAD_VERSION:   return "unsupported segment version";
    case DLSM_SHM_E_OOM:           return "arena exhausted";
    case DLSM_SHM_E_INVAL:         return "invalid argument";
    default:                       return "unknown error";
    }
}

dlsm_status dlsm_shm_create_or_recover(const char *name, size_t size, dlsm_shm **out) {
    if (!name || !out || size < sizeof(struct ShmHeader) + 16) {
        return DLSM_SHM_E_INVAL;
    }
    int fd = shm_open(name, O_CREAT | O_EXCL | O_RDWR, 0600);
    if (fd < 0) {
        if (errno == EEXIST) {
            return DLSM_SHM_E_IN_USE; /* recovery path filled in Task 5 */
        }
        return DLSM_SHM_E_OPEN;
    }
    if (ftruncate(fd, (off_t)size) != 0) {
        close(fd);
        shm_unlink(name);
        return DLSM_SHM_E_FTRUNCATE;
    }
    void *p = mmap((void *)(uintptr_t)DLSM_SHM_BASE_ADDR, size,
                   PROT_READ | PROT_WRITE,
                   MAP_SHARED | MAP_FIXED_NOREPLACE, fd, 0);
    if (p == MAP_FAILED || p != (void *)(uintptr_t)DLSM_SHM_BASE_ADDR) {
        if (p != MAP_FAILED) { munmap(p, size); }
        close(fd);
        shm_unlink(name);
        return DLSM_SHM_E_BASE_OCCUPIED;
    }
    init_header((struct ShmHeader *)p, size);

    dlsm_shm *s = calloc(1, sizeof(*s));
    if (!s) { munmap(p, size); close(fd); shm_unlink(name); return DLSM_SHM_E_INVAL; }
    s->hdr = (struct ShmHeader *)p;
    s->mapped_size = size;
    s->fd = fd;
    s->readonly = false;
    *out = s;
    return DLSM_OK;
}

void dlsm_shm_detach(dlsm_shm *s) {
    if (!s) { return; }
    munmap(s->hdr, s->mapped_size);
    close(s->fd);
    free(s);
}

void  *dlsm_shm_base(const dlsm_shm *s)     { return s->hdr; }
size_t dlsm_shm_capacity(const dlsm_shm *s) { return (size_t)s->hdr->total_size; }
size_t dlsm_shm_used(const dlsm_shm *s) {
    return (size_t)atomic_load_explicit(&s->hdr->bump, memory_order_acquire);
}

/* Minimal cleanup: unconditional unlink. Task 5 replaces with liveness check. */
dlsm_status dlsm_shm_cleanup_if_stale(const char *name, bool *cleaned) {
    int rc = shm_unlink(name);
    if (cleaned) { *cleaned = (rc == 0); }
    return DLSM_OK;
```

```c
#include <pthread.h>

/* xorshift64* — deterministic, seedable. */
typedef struct { uint64_t s; } rng_t;
static inline void rng_seed(rng_t *r, uint64_t seed) { r->s = seed ? seed : 0x9E3779B97F4A7C15ULL; }
static inline uint64_t rng_next(rng_t *r) {
    uint64_t x = r->s;
    x ^= x >> 12; x ^= x << 25; x ^= x >> 27;
    r->s = x;
    return x * 0x2545F4914F6CDD1DULL;
}
/* uniform in [lo, hi] inclusive */
static inline uint64_t rng_range(rng_t *r, uint64_t lo, uint64_t hi) {
    return lo + (rng_next(r) % (hi - lo + 1));
}

/* Start n threads on fn, wait for all. */
static inline void thread_team(int n, void *(*fn)(void *), void **args) {
    pthread_t th[64];
    for (int i = 0; i < n; i++) { pthread_create(&th[i], NULL, fn, args[i]); }
```

```c
static int is_pow2(size_t a) { return a != 0 && (a & (a - 1)) == 0; }

void *dlsm_shm_alloc(dlsm_shm *s, size_t size, size_t align) {
    if (!s || s->readonly || size == 0 || !is_pow2(align)) {
        return NULL;
    }
    struct ShmHeader *h = s->hdr;
    uint64_t cur = atomic_load_explicit(&h->bump, memory_order_acquire);
    for (;;) {
        uint64_t aligned = (cur + (align - 1)) & ~((uint64_t)align - 1);
        uint64_t next = aligned + size;
        if (next > h->total_size) {
            return NULL; /* E_OOM */
        }
        if (atomic_compare_exchange_weak_explicit(
                &h->bump, &cur, next,
                memory_order_acq_rel, memory_order_acquire)) {
            return (uint8_t *)h + aligned;
        }
        /* cur reloaded by CAS on failure; retry */
    }
}
```

```c
/* Map the header read-only to learn base/size/owner. Returns 0 on success and
 * fills *out_owner/*out_size; sets *bad to an shm status on magic/version error. */
static int read_header(int fd, int32_t *out_owner, uint64_t *out_size, dlsm_status *bad) {
    struct ShmHeader tmp_storage;
    void *t = mmap(NULL, sizeof(struct ShmHeader), PROT_READ, MAP_SHARED, fd, 0);
    if (t == MAP_FAILED) { *bad = DLSM_SHM_E_OPEN; return -1; }
    memcpy(&tmp_storage, t, sizeof(tmp_storage));
    munmap(t, sizeof(struct ShmHeader));
    if (tmp_storage.magic != DLSM_SHM_MAGIC)      { *bad = DLSM_SHM_E_BAD_MAGIC;   return -1; }
    if (tmp_storage.version != DLSM_SHM_VERSION)  { *bad = DLSM_SHM_E_BAD_VERSION; return -1; }
    *out_owner = tmp_storage.owner_pid;
    *out_size  = tmp_storage.total_size;
    return 0;
}

/* 1 if pid is alive (or we lack permission to know), 0 if definitely dead. */
static int pid_alive(int32_t pid) {
    if (kill((pid_t)pid, 0) == 0) { return 1; }
    return (errno == EPERM) ? 1 : 0; /* ESRCH -> dead */
```

```c
    if (!s) { munmap(p, (size_t)hsize); close(fd); return DLSM_SHM_E_INVAL; }
    s->hdr = (struct ShmHeader *)p;
    s->mapped_size = (size_t)hsize;
    s->fd = fd;
    s->readonly = true;
    *out = s;
    return DLSM_OK;
}
```

## Task 7: Concurrency stress (L3/L4) under TSAN

**Files:**
- Create: `libs/shm/tests/stress_shm_alloc.c`
- Modify: `libs/shm/CMakeLists.txt`

- [ ] **Step 1: Write `libs/shm/tests/stress_shm_alloc.c`**

Verifies invariant I3 (concurrent alloc, no overlap) by having N threads record every pointer they get, then the main thread sorts all pointers and asserts no two allocations overlap and all are in-region/aligned.

`libs/shm/tests/stress_shm_alloc.c`:
```c
#include "dlsm/shm.h"
#include "test_util.h"
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <string.h>

#define NAME "/dlsm_stress_alloc"
#define SIZE (64u << 20)        /* 64 MiB */
#define NTHREAD 8
#define PER_THREAD 50000
#define BLK 64                  /* fixed block size for overlap check */

typedef struct { dlsm_shm *s; uintptr_t *out; int n; } job_t;

static void *worker(void *arg) {
    job_t *j = (job_t *)arg;
    for (int i = 0; i < j->n; i++) {
        void *p = dlsm_shm_alloc(j->s, BLK, 16);
        if (!p) { j->out[i] = 0; continue; }
        j->out[i] = (uintptr_t)p;
    }
    return NULL;
}

static int cmp_uptr(const void *a, const void *b) {
    uintptr_t x = *(const uintptr_t *)a, y = *(const uintptr_t *)b;
    return (x > y) - (x < y);
}

int main(void) {
    dlsm_shm_cleanup_if_stale(NAME, NULL);
    dlsm_shm *s = NULL;
    if (dlsm_shm_create_or_recover(NAME, SIZE, &s) != DLSM_OK) {
        fprintf(stderr, "create failed\n"); return 1;
    }
    static uintptr_t bufs[NTHREAD][PER_THREAD];
    job_t jobs[NTHREAD];
    void *args[NTHREAD];
    for (int i = 0; i < NTHREAD; i++) {
        jobs[i] = (job_t){ .s = s, .out = bufs[i], .n = PER_THREAD };
        args[i] = &jobs[i];
    }
    thread_team(NTHREAD, worker, args);

    /* gather non-null pointers */
    static uintptr_t all[NTHREAD * PER_THREAD];
    int n = 0;
    uintptr_t base = (uintptr_t)dlsm_shm_base(s);
    uintptr_t end  = base + dlsm_shm_capacity(s);
    for (int t = 0; t < NTHREAD; t++) {
        for (int i = 0; i < PER_THREAD; i++) {
            uintptr_t p = bufs[t][i];
            if (!p) { continue; }
            if (p < base || p + BLK > end || (p % 16) != 0) {
                fprintf(stderr, "invariant violated: bad ptr %p\n", (void *)p);
                return 2;
            }
            all[n++] = p;
        }
    }
    qsort(all, n, sizeof(uintptr_t), cmp_uptr);
    for (int i = 1; i < n; i++) {
        if (all[i] < all[i - 1] + BLK) {
            fprintf(stderr, "overlap: %p and %p\n", (void *)all[i-1], (void *)all[i]);
            return 3;
        }
    }
    dlsm_shm_detach(s);
    dlsm_shm_cleanup_if_stale(NAME, NULL);
    printf("stress ok: %d allocations, no overlap\n", n);
    return 0;
}
```

```c
#include "dlsm/shm.h"
#include <stdint.h>
#include <stddef.h>
#include <assert.h>

#define NAME "/dlsm_fuzz_alloc"
#define SIZE (1u << 20)

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t len) {
    dlsm_shm_cleanup_if_stale(NAME, NULL);
    dlsm_shm *s = NULL;
    if (dlsm_shm_create_or_recover(NAME, SIZE, &s) != DLSM_OK) { return 0; }
    uintptr_t base = (uintptr_t)dlsm_shm_base(s);
    uintptr_t end  = base + dlsm_shm_capacity(s);
    size_t i = 0;
    while (i + 2 <= len) {
        size_t sz = (size_t)data[i] + 1;                 /* 1..256 */
        size_t align = (size_t)1u << (data[i + 1] & 7);  /* 1,2,...,128 */
        i += 2;
        void *p = dlsm_shm_alloc(s, sz, align);
        if (p) {
            uintptr_t x = (uintptr_t)p;
            assert(x >= base && x + sz <= end);
            assert((x % align) == 0);
        }
    }
    dlsm_shm_detach(s);
    dlsm_shm_cleanup_if_stale(NAME, NULL);
    return 0;
}
```

**Files:**

- [ ] **Step 1: Write `libs/shm/bench/bench_shm_alloc.c`**

`libs/shm/bench/bench_shm_alloc.c`:
```c
#include "dlsm/shm.h"
#include "../tests/test_util.h"
#include <stdio.h>
#include <stdlib.h>
#include <time.h>
#include <stdint.h>

#define NAME "/dlsm_bench_alloc"
#define SIZE (1024u << 20)   /* 1 GiB arena */
#define NTHREAD 8

typedef struct { dlsm_shm *s; long ops; } bjob_t;

static void *bworker(void *arg) {
    bjob_t *j = (bjob_t *)arg;
    long n = 0;
    while (dlsm_shm_alloc(j->s, 16, 16)) { n++; }
    j->ops = n;
    return NULL;
}

int main(void) {
    dlsm_shm_cleanup_if_stale(NAME, NULL);
    dlsm_shm *s = NULL;
    if (dlsm_shm_create_or_recover(NAME, SIZE, &s) != DLSM_OK) {
        fprintf(stderr, "create failed (need ~1GiB shm)\n"); return 1;
    }
    bjob_t jobs[NTHREAD]; void *args[NTHREAD];
    for (int i = 0; i < NTHREAD; i++) { jobs[i] = (bjob_t){ .s = s, .ops = 0 }; args[i] = &jobs[i]; }

    struct timespec t0, t1;
    clock_gettime(CLOCK_MONOTONIC, &t0);
    thread_team(NTHREAD, bworker, args);
    clock_gettime(CLOCK_MONOTONIC, &t1);

    long total = 0;
    for (int i = 0; i < NTHREAD; i++) { total += jobs[i].ops; }
    double secs = (t1.tv_sec - t0.tv_sec) + (t1.tv_nsec - t0.tv_nsec) / 1e9;
    double mops = total / secs / 1e6;
    printf("alloc: %ld ops in %.3fs = %.1f Mops/s (%d threads)\n", total, secs, mops, NTHREAD);

    dlsm_shm_detach(s);
    dlsm_shm_cleanup_if_stale(NAME, NULL);

    /* DoD threshold: > 50 Mops/s. Allow override for slow CI via env. */
    const char *floor_s = getenv("DLSM_BENCH_FLOOR_MOPS");
    double floor = floor_s ? atof(floor_s) : 50.0;
    if (mops < floor) { fprintf(stderr, "below floor %.1f Mops/s\n", floor); return 2; }
    return 0;
}
```
